            } catch (error) {
                if (error.name !== 'AbortError') {
                    console.error(error);
                    this.results = [];
                    this.loading = false;
                }
            }
        },
        
        select(item) {
            this.selectedNim = item.id;
            this.query = item.text;
            this.open = false;
        }
    }
}
